Overview

Filgrastim (G-CSF) PK/PD design evaluation with PFIM: three parallel dose groups (1, 3, 10 mg/kg SC every 24 h for 7 doses, body weight 75 kg, 10 subjects per group), dense PK (CP) and ANC (NB) sampling over 10 days (Krzyzanski et al. 2010).

To reproduce this workflow interactively in RStudio:

  1. Set the working directory to the PFIM package root (folder with DESCRIPTION).
  2. Run devtools::load_all(".") — not library(PFIM) alone (that loads an older installed build).
  3. Do not re-use an evaluation_gcsf object from a previous session. Either run all code chunks below in order, or in one step:
devtools::load_all(".")
source("tools/run_gcsf_evaluation.R")  # creates evaluation_gcsf + runs FIM

Quick check: after run(), det(FIM) > 0 and getRSE(evaluation_gcsf)["μ_NB0", "RSE"] should be about 11%, not ~1000%. If you only see 8 fixed effects (μ_KA…μ_SM2 without μ_KD, μ_KINT, μ_KMT, μ_KTT), you are still on the old parameterization.

The hidden defs chunk runs automatically when knitting only.

Model equations

# Baseline steady-state block (MAIN): derived from primary parameters so that
# gradients w.r.t. NB0, KMT, etc. stay consistent (as in mrgsolve/PopED).
gcsf_main_block = paste(
  "BAS = 0.0246; CP0 = BAS;",
  "H10 = 1+SM1*CP0/(SC1+CP0); H20 = 1+SM2*CP0/(SC1+CP0); H30 = 1+SM3*CP0/(SC1+CP0);",
  "KINB = KMT*NB0/H10;",
  "den = KBB1*H30 + KTT*H20;",
  "BM10 = KINB*H10/den; BM20 = KTT*H20*BM10/den;",
  "BM30 = KTT*H20*BM20/den; BM40 = KTT*H20*BM30/den;",
  "BM50 = KTT*H20*BM40/den; BM60 = KTT*H20*BM50/den;",
  "BM70 = KTT*H20*BM60/den; BM80 = KTT*H20*BM70/den; BM90 = KTT*H20*BM80/den;",
  "NT0 = BM10+BM20+BM30+BM40+BM50+BM60+BM70+BM80+BM90+NB0;",
  "AC0 = CP0*VD; ADR0 = KSI*NT0*AC0/(KD+CP0);",
  "KIN = KEL*AC0 + KINT*ADR0;"
)

gcsf_ode_prefix = paste(
  gcsf_main_block,
  "ATOT = CENT;",
  "NT = B1+B2+B3+B4+B5+B6+B7+B8+B9+NB; RTOT = KSI*NT;",
  "BB = RTOT-ATOT/VD+KD; CP = 0.5*(-BB+sqrt(BB^2+4*KD*ATOT/VD));",
  "AC = CP*VD; ADR = RTOT*AC/(KD+CP);",
  "H1 = 1+SM1*CP/(SC1+CP); H2 = 1+SM2*CP/(SC1+CP); H3 = 1+SM3*CP/(SC1+CP);"
)

modelEquations = list(
  Deriv_ABS  = paste0("{ ", gcsf_ode_prefix, " -KA*ABS }"),
  Deriv_CENT = "{ KA*ABS + KIN - KEL*AC - KINT*ADR }",
  Deriv_FRR  = "0", Deriv_DR = "0",
  Deriv_B1   = "{ KINB*H1*(B1/BM10) - KBB1*H3*B1 - KTT*H2*B1 }",
  Deriv_B2   = "{ KTT*H2*B1 - KBB1*H3*B2 - KTT*H2*B2 }",
  Deriv_B3   = "{ KTT*H2*B2 - KBB1*H3*B3 - KTT*H2*B3 }",
  Deriv_B4   = "{ KTT*H2*B3 - KBB1*H3*B4 - KTT*H2*B4 }",
  Deriv_B5   = "{ KTT*H2*B4 - KBB1*H3*B5 - KTT*H2*B5 }",
  Deriv_B6   = "{ KTT*H2*B5 - KBB1*H3*B6 - KTT*H2*B6 }",
  Deriv_B7   = "{ KTT*H2*B6 - KBB1*H3*B7 - KTT*H2*B7 }",
  Deriv_B8   = "{ KTT*H2*B7 - KBB1*H3*B8 - KTT*H2*B8 }",
  Deriv_B9   = "{ KTT*H2*B8 - KBB1*H3*B9 - KTT*H2*B9 }",
  Deriv_NB   = "{ KTT*H2*B9 + KBB1*H3*(B1+B2+B3+B4+B5+B6+B7+B8+B9) - KMT*NB }"
)

outputs = list(
  RespPK = paste0(
    "0.5*(-(KSI*(B1+B2+B3+B4+B5+B6+B7+B8+B9+NB)-CENT/VD+KD)",
    "+sqrt((KSI*(B1+B2+B3+B4+B5+B6+B7+B8+B9+NB)-CENT/VD+KD)^2+4*KD*CENT/VD))"
  ),
  RespPD = "NB"
)

Parameters, error model, initial conditions

# DDMORE population mean (Krzyzanski et al.)
p = c(FF=0.626, KA=0.642, KINT=0.101, KSI=0.211, KMT=0.0723, KBB1=0, KTT=0.0102,
       KEL=0.148, VD=2.56, KD=1.27, NB0=1.65, SC1=3.21, SM1=34.3, SM2=32.3, SM3=0)

# Primary parameters only (KINB, BM10, KIN are derived in gcsf_main_block).
# KD, KINT, KMT, KTT are estimated fixed effects in PopED (no IIV).
modelParameters = list(
  ModelParameter(name="KA",   distribution=LogNormal(mu=p["KA"],   omega=0)),
  ModelParameter(name="KEL",  distribution=LogNormal(mu=p["KEL"],  omega=sqrt(0.312))),
  ModelParameter(name="VD",   distribution=LogNormal(mu=p["VD"],   omega=sqrt(0.328))),
  ModelParameter(name="KD",   distribution=LogNormal(mu=p["KD"],   omega=0)),
  ModelParameter(name="KINT", distribution=LogNormal(mu=p["KINT"], omega=0)),
  ModelParameter(name="KSI",  distribution=LogNormal(mu=p["KSI"],  omega=sqrt(0.224))),
  ModelParameter(name="KMT",  distribution=LogNormal(mu=p["KMT"],  omega=0)),
  ModelParameter(name="KTT",  distribution=LogNormal(mu=p["KTT"],  omega=0)),
  ModelParameter(name="NB0",  distribution=LogNormal(mu=p["NB0"],  omega=sqrt(0.298))),
  ModelParameter(name="SC1",  distribution=LogNormal(mu=p["SC1"],  omega=sqrt(0.803))),
  ModelParameter(name="SM1",  distribution=LogNormal(mu=p["SM1"],  omega=sqrt(0.0128))),
  ModelParameter(name="SM2",  distribution=LogNormal(mu=p["SM2"],  omega=0)),
  ModelParameter(name="SM3",  distribution=LogNormal(mu=p["SM3"],  omega=0), fixedMu=TRUE),
  ModelParameter(name="FF",   distribution=LogNormal(mu=p["FF"],   omega=0), fixedMu=TRUE),
  ModelParameter(name="KBB1", distribution=LogNormal(mu=p["KBB1"],omega=0), fixedMu=TRUE)
)

modelError = list(
  Combined1(output="RespPK", sigmaInter=0,    sigmaSlope=0.253),
  Combined1(output="RespPD", sigmaInter=2.10, sigmaSlope=0.0227)
)

# String expressions: re-evaluated from current mu on each gradient step.
gcsf_baseline_ic = function(state) {
  blk = paste(
    "local({ BAS = 0.0246; CP0 = BAS;",
    "H10 = 1+SM1*CP0/(SC1+CP0); H20 = 1+SM2*CP0/(SC1+CP0); H30 = 1+SM3*CP0/(SC1+CP0);",
    "KINB = KMT*NB0/H10; den = KBB1*H30 + KTT*H20;",
    "BM10 = KINB*H10/den; BM20 = KTT*H20*BM10/den;",
    "BM30 = KTT*H20*BM20/den; BM40 = KTT*H20*BM30/den;",
    "BM50 = KTT*H20*BM40/den; BM60 = KTT*H20*BM50/den;",
    "BM70 = KTT*H20*BM60/den; BM80 = KTT*H20*BM70/den; BM90 = KTT*H20*BM80/den;",
    "NT0 = BM10+BM20+BM30+BM40+BM50+BM60+BM70+BM80+BM90+NB0;",
    "AC0 = CP0*VD; ADR0 = KSI*NT0*AC0/(KD+CP0);"
  )
  paste0(blk, " ", state, " })")
}

# Order must match modelEquations (ABS, CENT, FRR, DR, B1–B9, NB) for deSolve.
initialConditions = list(
  ABS = 0,
  CENT = gcsf_baseline_ic("AC0+ADR0"),
  FRR = 0,
  DR = 0,
  B1 = gcsf_baseline_ic("BM10"), B2 = gcsf_baseline_ic("BM20"), B3 = gcsf_baseline_ic("BM30"),
  B4 = gcsf_baseline_ic("BM40"), B5 = gcsf_baseline_ic("BM50"), B6 = gcsf_baseline_ic("BM60"),
  B7 = gcsf_baseline_ic("BM70"), B8 = gcsf_baseline_ic("BM80"), B9 = gcsf_baseline_ic("BM90"),
  NB = "NB0"
)

Design

pk_dense = c(0.167, 0.25, 0.333, 0.5, 0.667, 0.75, 0.833, 1, 1.5, 2, 3, 4, 5,
              6, 8, 10, 12, 14, 16, 18, 20, 24)
anc_dense = c(0.333, 0.5, 0.667, 0.75, 1, 1.5, 2, 3, 4, 5, 6, 8, 10, 12, 14,
               16, 18, 20, 24)
pk_times  = c(pk_dense, pk_dense + 6 * 24, 7 * 24 + c(4, 24, 48))
anc_times = c(anc_dense, (2:6) * 24, anc_dense + 6 * 24, 7 * 24 + c(4, 24, 48, 72))

gcsf_arm = function(dose_mgkg, group = 1L) {
  Arm(
    name = paste0("Group ", group),
    size = 10,
    administrations = list(
      Administration(
        outcome = "ABS",
        timeDose = seq(0, 6 * 24, 24),
        dose = rep(dose_mgkg * 75 * as.numeric(p["FF"]), 7),
        tau = 0
      )
    ),
    samplingTimes = list(
      SamplingTimes(outcome = "RespPK", samplings = pk_times),
      SamplingTimes(outcome = "RespPD", samplings = anc_times)
    ),
    initialConditions = initialConditions
  )
}

design_gcsf = Design(
  name = "gcsf_design",
  arms = list(gcsf_arm(1, 1L), gcsf_arm(3, 2L), gcsf_arm(10, 3L))
)

evaluation_gcsf = Evaluation(
  name = "G-CSF",
  modelEquations = modelEquations,
  modelParameters = modelParameters,
  modelError = modelError,
  outputs = outputs,
  designs = list(design_gcsf),
  fimType = "population",
  odeSolverParameters = list(atol = 1e-8, rtol = 1e-8)
)

Test plot

PK and ANC profiles for groups 1, 3 and 10 mg/kg (same layout as the optimal-design tutorial):

plot_gcsf_test = function(evaluation, n_points = 200) {
  model = defineModelWrapper(defineModelType(evaluation), evaluation)
  arms  = S7::prop(S7::prop(evaluation, "designs")[[1]], "arms")
  times = seq(0, 240, length.out = n_points)

  curves = purrr::list_rbind(purrr::map(seq_along(arms), function(i) {
    plot_arm = arms[[i]]
    S7::prop(plot_arm, "samplingTimes") = list(
      SamplingTimes("RespPK", times),
      SamplingTimes("RespPD", times)
    )
    m = defineModelAdministration(model, plot_arm)
    y = evaluateModel(m, plot_arm)
    rbind(
      data.frame(time = y$RespPK$time, value = y$RespPK$RespPK, model = "PK",  group = i),
      data.frame(time = y$RespPD$time, value = y$RespPD$RespPD, model = "ANC", group = i)
    )
  }))

  samp_pts = purrr::list_rbind(purrr::map(seq_along(arms), function(i) {
    samp = S7::prop(arms[[i]], "samplingTimes")
    rbind(
      data.frame(time = S7::prop(samp[[1]], "samplings"), model = "PK",  group = i),
      data.frame(time = S7::prop(samp[[2]], "samplings"), model = "ANC", group = i)
    )
  }))

  pts = purrr::list_rbind(purrr::map(seq_len(nrow(samp_pts)), function(i) {
    row = samp_pts[i, ]
    sub = curves[curves$model == row$model & curves$group == row$group, ]
    j   = which.min(abs(sub$time - row$time))
    data.frame(time = row$time, value = sub$value[j], model = row$model, group = row$group)
  }))

  ggplot(curves, aes(time, value, color = factor(group))) +
    geom_line(linewidth = 0.9) +
    geom_point(data = pts, aes(time, value), size = 2.5) +
    facet_wrap(~model, scales = "free_y") +
    scale_color_discrete(name = "Group", labels = c("1", "2", "3")) +
    labs(
      x = "Time from first dose (hours)",
      y = "Model predictions",
      title = "Test plot"
    )
}

plot_gcsf_test(evaluation_gcsf)

Agreement: fixed effects NB0, KEL, VD, SC1, SM1, SM2 and most IIV terms (ω²_KEL, ω²_VD, ω²_SC1, ω²_SM1) are within about 15% of PopED.

Larger gaps: KA, KD, KINT, ω²_NB0, and residual-error RSE reflect differences between PopED’s mrgsolve-based FIM and PFIM’s finite-difference FIM on this large ODE system — not a singular FIM (getDeterminant(evaluation_gcsf) > 0) or a mismatched design.

Compare candidate dose sets

candidate_doses = list(
  `Design 1` = c(1, 3, 10),
  `Design 2` = c(0.3, 1, 3),
  `Design 3` = c(0.3, 1, 10),
  `Design 4` = c(0.1, 1, 3)
)

purrr::imap(candidate_doses, function(doses, label) {
  d = Design(name = label, arms = purrr::imap(doses, ~ gcsf_arm(.x, .y)))
  ev = Evaluation(
    name = label,
    modelEquations = modelEquations,
    modelParameters = modelParameters,
    modelError = modelError,
    outputs = outputs,
    designs = list(d),
    fimType = "population",
    odeSolverParameters = list(atol = 1e-6, rtol = 1e-6)
  )
  c(design = label, t(round(getRSE(run(ev)), 2)))
})
